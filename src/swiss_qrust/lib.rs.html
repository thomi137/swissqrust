7//! Generate Swiss QR-bill payment slips (ISO 20022 / SIX Implementation
8//! Guidelines) as PDF or SVG.
9//!
10//! The typical flow is: parse an [`InputBill`] from TOML or JSON, convert it
11//! to a validated [`BillData`] via [`TryFrom`], then render it.
12//!
13//! ```
14//! use swiss_qrust::{BillData, InputBill, Language};
15//!
16//! let toml = r#"
17//! iban = "CH93 0076 2011 6238 5295 7"
18//! currency = "CHF"
19//! amount = "199.95"
20//!
21//! [creditor_address]
22//! name = "Robert Schneider AG"
23//! street = "Rue du Lac"
24//! house_num = "1268"
25//! plz = "2501"
26//! city = "Biel"
27//! country = "CH"
28//! "#;
29//!
30//! let input: InputBill = toml::from_str(toml)?;
31//! let bill = BillData::try_from(input)?;
32//!
33//! let svg = swiss_qrust::svg::render_bill_to_svg(&bill, Language::De)?;
34//! assert!(svg.starts_with("<svg"));
35//! # Ok::<(), anyhow::Error>(())
36//! ```
37//!
38//! For a PDF instead of SVG, use [`pdf::render_bill_to_pdf`] (returns the
39//! raw bytes) or [`pdf::create_pdf`] (writes straight to a file). Both
40//! render engines consume the same [`BillData`] - there is nothing
41//! PDF-specific or SVG-specific about validating or building the bill
42//! itself.
43//!
44//! [`InputBill`]: input::InputBill
45//! [`BillData`]: bill::BillData

74/// Parses `content` as an [`InputBill`], choosing the format by `extension`.
75///
76/// `extension` is matched literally against `"toml"` or `"json"` (as you'd
77/// get from `Path::extension()`); anything else is an error. This is a thin
78/// convenience wrapper - if you already know the format, call
79/// `toml::from_str`/`serde_json::from_str` directly instead.
80///
81/// ```
82/// use swiss_qrust::parse_bill_data;
83///
84/// let toml = r#"
85/// iban = "CH93 0076 2011 6238 5295 7"
86/// currency = "CHF"
87///
88/// [creditor_address]
89/// name = "Robert Schneider AG"
90/// street = "Rue du Lac"
91/// house_num = "1268"
92/// plz = "2501"
93/// city = "Biel"
94/// country = "CH"
95/// "#;
96///
97/// let bill = parse_bill_data(toml, "toml")?;
98/// assert_eq!(bill.iban, "CH93 0076 2011 6238 5295 7");
99/// # Ok::<(), anyhow::Error>(())
100/// ```
101pub fn parse_bill_data(content: &str, extension: &str) -> anyhow::Result<InputBill> {
102    match extension {
103        "toml" => Ok(toml::from_str(content)?),
104        "json" => Ok(serde_json::from_str(content)?),
105        other => Err(anyhow::anyhow!("Unsupported input format: {other}")),
106    }
107}
