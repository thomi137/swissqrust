17/// Handles SVG rendering operations
18pub fn execute_bill_ops_svg (
19    ops: Vec<DrawOp>,
20    qr_data: Option<&QrCode>,
21) -> Result<String, RenderError> {
22
23    // 1. Prepare the Data URIs
24    let reg_uri = get_font_data_uri(LIBERATION_SANS_REGULAR_TTF);
25    let bold_uri = get_font_data_uri(LIBERATION_SANS_BOLD_TTF);
26
27    // 2. Create the CSS string
28    let font_css = format!(r#"
29        @font-face {{
30            font-family: 'LiberationSans';
31            src: url('{}') format('truetype');
32            font-weight: normal;
33            font-style: normal;
34        }}
35        @font-face {{
36            font-family: 'LiberationSans';
37            src: url('{}') format('truetype');
38            font-weight: bold;
39            font-style: normal;
40        }}
41    "#, reg_uri, bold_uri);
42
43    let y_offset = 192.0;
44
45    let mut doc = Document::new()
46        .set("viewBox", (0, 0, 210, 105))
47        .set("width", "100%")
48        .set("height", "100%")
49        .set("preserveAspectRatio", "xMidYMid meet")
50        .set("class", "swiss-qr-preview");
51
52    for op in ops.iter() {
53        match op {
54            DrawOp::Text { text, at, size, bold } => {
55                // The slip starts at 192mm from the top of an A4 page (297 - 105)
56                let svg_y = at.y.0.0 - y_offset;
57                let txt = SvgText::new(text)
58                    .set("x", at.x.0)
59                    .set("y", svg_y) // SVG is naturally top-left!
60                    .set("font-size", &size.0 * MM_PER_PT) // viewport units (mm in this case)
61                    .set("font-family", "LiberationSans, Arial, sans-serif")
62                    .set("font-weight", if *bold { "bold" } else { "normal" });
63                doc = doc.add(txt)
64            },
65            DrawOp::Line { from, to, width } => {
66                let line = SvgLine::new()
67                    .set("x1", from.0 .0)
68                    .set("y1", from.1.0.0 -  y_offset)
69                    .set("x2", to.0 .0)
70                    .set("y2", to.1.0.0 - y_offset)
71                    .set("stroke", "black")
72                    .set("stroke-width", width.0);
73                doc = doc.add(line);
74            },
75            DrawOp::Box { rect } => {
76                let box_rect = Rectangle::new()
77                    .set("x", rect.x.0)
78                    .set("y", rect.y.0 - y_offset)
79                    .set("width", rect.width.0)
80                    .set("height", rect.height.0)
81                    .set("fill", "none")
82                    .set("stroke", "black")
83                    .set("stroke-width", 0.2); // 0.75pt approx
84                doc = doc.add(box_rect);
85            },
86            DrawOp::QrCodeSpace { at, .. } => {
87                if let Some(qr) = qr_data {
88                    let y_svg = 17f32;
89                    let qr_group = add_swiss_cross(
90                        render_qr_svg(qr.clone(), at.x.0 as f64, y_svg as f64)
91                    );
92                    doc = doc.add(qr_group);
93                }
94            },
95        }
96    }
97
98    doc = add_perforation_marks(doc);
99
100    Ok(doc.to_string())
101}

103fn get_font_data_uri(bytes: &[u8]) -> String {
104    let b64 = general_purpose::STANDARD.encode(bytes);
105    format!("data:font/ttf;base64,{}", b64)
106}
