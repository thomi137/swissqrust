6use svg::node::element::Line as SvgLine;
7use svg::Document;
8
9/// Style Guide p.7: on the printed/PDF output, the QR-bill is separated from
10/// the rest of the document by a dashed perforation line with a scissors
11/// symbol (ZapfDingbats), both along its top edge and between receipt and
12/// payment part. Those are print-cut marks, not meaningful in an on-screen
13/// preview - the PDF engine (`pdf_helpers.rs`) already draws them for the
14/// actual output. Here we only keep a plain divider between receipt and
15/// payment part, to visually separate the two without implying a real
16/// perforation/cut mark exists on screen.
17pub fn add_perforation_marks(mut doc: Document) -> Document {
18    let vertical = SvgLine::new()
19        .set("x1", 62)
20        .set("y1", 0)
21        .set("x2", 62)
22        .set("y2", 105)
23        .set("stroke", "black")
24        .set("stroke-width", 0.26)
25        .set("stroke-dasharray", "1.06,1.06");
26    doc = doc.add(vertical);
27
28    doc
29}
