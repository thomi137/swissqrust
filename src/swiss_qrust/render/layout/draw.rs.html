13/// Truncates `text` to fit `available_width`, appending "…" if it had to be
14/// cut. Per Style Guide 3.5.4/3.5.5, overflowing text on the payment part
15/// must be marked with a trailing ellipsis rather than silently clipped.
16pub fn truncate_to_width<T: FontMetrics>(
17    metrics: &T,
18    text: &str,
19    style: FontStyle,
20    size: Pt,
21    available_width: Mm,
22) -> String {
23    if metrics.text_width_mm(text, style, size).0 <= available_width.0 {
24        return text.to_string();
25    }
26    let mut chars: Vec<char> = text.chars().collect();
27    while !chars.is_empty() {
28        chars.pop();
29        let candidate: String = chars.iter().collect::<String>() + "…";
30        if metrics.text_width_mm(&candidate, style, size).0 <= available_width.0 {
31            return candidate;
32        }
33    }
34    "…".to_string()
35}

37/// Draws a label and moves cursor down by line_spacing.
38pub fn draw_label(
39    ops: &mut Vec<DrawOp>,
40    text: &str,
41    x: Mm,
42    y: &mut Mm,
43    font_size: Pt,
44) {
45    ops.push(DrawOp::Text {
46        text: text.into(),
47        at: Baseline { x, y: LayoutY(*y) },
48        size: font_size,
49        bold: true,
50    });
51
52   // y.0 += line_spacing.0;
53}

71/// Draws multiple lines.
72pub fn draw_text_lines(
73    ops: &mut Vec<DrawOp>,
74    lines: &[String],
75    x: Mm,
76    y: &mut Mm,
77    font_size: Pt,
78    line_spacing: Mm,
79) {
80    for line in lines {
81        ops.push(DrawOp::Text {
82            text: line.clone(),
83            at: Baseline { x, y: LayoutY(*y) },
84            size: font_size,
85            bold: false,
86        });
87
88        *y = Mm(y.0 + line_spacing.0);
89    }
90
91   // *y = Mm(y.0 + extra_spacing.0);
92}

135/// Points-to-millimetre factor (1 pt = 1/72 inch). The corner-mark source
136/// artwork is authored in points at its native (receipt-sized) box; the
137/// Style Guide (p. 7) fixes the mark itself at 3 x 3 mm / 0.75 pt regardless
138/// of the surrounding placeholder box's dimensions, so this factor - not a
139/// per-box scale - is what converts the artwork to millimetres.
140const POINTS_TO_MM: f32 = 25.4 / 72.0;
141
142/// Draws the four Eckmarken (corner marks) for an empty placeholder box.
143///
144/// Each of the four polylines in `polylines` represents one L-shaped mark,
145/// anchored to one corner of `viewbox`. Marks are re-anchored to the
146/// matching corner of `rect` at their fixed native size (not stretched to
147/// fill `rect`), since a payment-part box (e.g. 65x25mm) and its receipt
148/// counterpart (52x20mm) must show identically-sized corner marks.
149pub fn draw_corner_marks(
150    ops: &mut Vec<DrawOp>,
151    rect: QRBillLayoutRect,
152    viewbox: (f64, f64),
153    polylines: &[Polygon]
154) {
155    let vb_w = viewbox.0 as f32;
156    let vb_h = viewbox.1 as f32;
157
158    for poly in polylines {
159        // Draw each polyline as a series of connected lines
160        if poly.points.len() < 2 {
161            continue;
162        }
163
164        // Every point of a given corner mark lies in the same quadrant of
165        // the source artwork, so the first point tells us which corner of
166        // `rect` this mark belongs to.
167        let (fx, fy) = (poly.points[0].0 as f32, poly.points[0].1 as f32);
168        let left = fx < vb_w / 2.0;
169        let top = fy < vb_h / 2.0;
170
171        let anchor = |px: f32, py: f32| -> (Mm, LayoutY) {
172            let off_x = if left { px } else { vb_w - px };
173            let off_y = if top { py } else { vb_h - py };
174            let mm_x = off_x * POINTS_TO_MM;
175            let mm_y = off_y * POINTS_TO_MM;
176
177            let x = if left { rect.x.0 + mm_x } else { rect.x.0 + rect.width.0 - mm_x };
178            let y = if top { rect.y.0 + rect.height.0 - mm_y } else { rect.y.0 + mm_y };
179            (Mm(x), LayoutY(Mm(y)))
180        };
181
182        for window in poly.points.windows(2) {
183            let (p1x, p1y) = (window[0].0 as f32, window[0].1 as f32);
184            let (p2x, p2y) = (window[1].0 as f32, window[1].1 as f32);
185
186            ops.push(DrawOp::Line {
187                from: anchor(p1x, p1y),
188                to: anchor(p2x, p2y),
189                width: Mm(0.21), // 0.75pt ≈ 0.21mm
190            });
191        }
192    }
193}
