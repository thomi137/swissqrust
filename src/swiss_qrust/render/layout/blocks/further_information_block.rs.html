7use crate::block_elements::{Column, ColumnCursor, LayoutBlock};
8use crate::constants::{A4_PAGE_HEIGHT, A4_PAGE_WIDTH, MARGIN, RECEIPT_WIDTH, PP_FURTHER_INFO_LINE_SPACING, PP_FURTHER_INFO_SECTION_TOP, PP_FURTHER_INFO_TEXT_SIZE};
9use crate::pdf::coords::LayoutY;
10use crate::render::FontMetrics;
11use crate::{truncate_to_width, Baseline, DrawOp, FontStyle, Mm, RenderContext};

13/// Style Guide 3.5.5 "Further information section" / "Weitere
14/// Informationen": up to two "Alternative procedures" lines at the bottom
15/// of the payment part, 7pt, with the (short) procedure name in bold. The
16/// receipt has no such section.
17pub struct FurtherInformationBlock;
18
19impl<T: FontMetrics> LayoutBlock<T> for FurtherInformationBlock {
20    fn column(&self) -> Column {
21        Column::Absolute
22    }
23
24    fn render(&self, ctx: &RenderContext<'_, T>, ops: &mut Vec<DrawOp>, _cursor: &mut ColumnCursor) {
25        let x = RECEIPT_WIDTH + MARGIN;
26        let available_width = A4_PAGE_WIDTH - x - MARGIN;
27        let row_top = A4_PAGE_HEIGHT - PP_FURTHER_INFO_SECTION_TOP
28            + ctx.metrics.ascender_mm(FontStyle::Bold, PP_FURTHER_INFO_TEXT_SIZE);
29
30        for (i, scheme) in ctx.bill_data.alternative_schemes.iter().flatten().enumerate() {
31            let y = row_top + Mm(i as f32) * PP_FURTHER_INFO_LINE_SPACING.to_mm();
32            draw_alt_procedure_line(ops, ctx, scheme, x, y, available_width);
33        }
34    }
35}
36
37/// Draws one Alternative-procedure line: the part up to and including the
38/// first separator ("eBill/" in "eBill/B/simon.muster@example.com") in
39/// bold, the rest in regular weight, truncated with "…" if it would
40/// otherwise overflow `available_width`.
41fn draw_alt_procedure_line<T: FontMetrics>(
42    ops: &mut Vec<DrawOp>,
43    ctx: &RenderContext<'_, T>,
44    scheme: &str,
45    x: Mm,
46    y: Mm,
47    available_width: Mm,
48) {
49    let (name, rest) = match scheme.find('/') {
50        Some(idx) => scheme.split_at(idx + 1),
51        None => (scheme, ""),
52    };
53
54    let name_width = ctx.metrics.text_width_mm(name, FontStyle::Bold, PP_FURTHER_INFO_TEXT_SIZE);
55    let rest = truncate_to_width(
56        ctx.metrics,
57        rest,
58        FontStyle::Regular,
59        PP_FURTHER_INFO_TEXT_SIZE,
60        available_width - name_width,
61    );
62
63    ops.push(DrawOp::Text {
64        text: name.to_string(),
65        at: Baseline { x, y: LayoutY(y) },
66        size: PP_FURTHER_INFO_TEXT_SIZE,
67        bold: true,
68    });
69
70    if !rest.is_empty() {
71        ops.push(DrawOp::Text {
72            text: rest,
73            at: Baseline { x: x + name_width, y: LayoutY(y) },
74            size: PP_FURTHER_INFO_TEXT_SIZE,
75            bold: false,
76        });
77    }
78}

87    fn ctx_and_fonts() -> (BillData, SvgFontLibrary) {
88        let toml = r#"
89iban = "CH93 0076 2011 6238 5295 7"
90currency = "CHF"
91
92[creditor_address]
93name = "Robert Schneider AG"
94street = "Rue du Lac"
95house_num = "1268"
96plz = "2501"
97city = "Biel"
98country = "CH"
99"#;
100        let input: InputBill = toml::from_str(toml).unwrap();
101        (BillData::try_from(input).unwrap(), SvgFontLibrary::new())
102    }

104    #[test]
105    fn splits_bold_name_from_regular_rest_at_first_slash() {
106        let (bill, fonts) = ctx_and_fonts();
107        let ctx = RenderContext::for_strategy::<PaymentPartLayout<SvgFontLibrary>>(&bill, Language::De, &fonts);
108
109        let mut ops = Vec::new();
110        draw_alt_procedure_line(&mut ops, &ctx, "eBill/B/simon.muster@example.com", Mm(0.0), Mm(0.0), Mm(100.0));
111
112        assert_eq!(ops.len(), 2, "expected a bold name segment and a regular rest segment");
113        let DrawOp::Text { text: name, bold: name_bold, .. } = &ops[0] else { panic!("expected Text") };
114        let DrawOp::Text { text: rest, bold: rest_bold, .. } = &ops[1] else { panic!("expected Text") };
115        assert_eq!(name, "eBill/");
116        assert!(name_bold);
117        assert_eq!(rest, "B/simon.muster@example.com");
118        assert!(!rest_bold);
119    }
120
121    #[test]
122    fn scheme_without_a_slash_is_drawn_as_bold_name_only() {
123        let (bill, fonts) = ctx_and_fonts();
124        let ctx = RenderContext::for_strategy::<PaymentPartLayout<SvgFontLibrary>>(&bill, Language::De, &fonts);
125
126        let mut ops = Vec::new();
127        draw_alt_procedure_line(&mut ops, &ctx, "NoSlashHere", Mm(0.0), Mm(0.0), Mm(100.0));
128
129        // No second (regular-weight) segment should be pushed when there's
130        // nothing left after the name - an empty DrawOp::Text would still
131        // "work" but is a wasted/misleading draw op.
132        assert_eq!(ops.len(), 1, "expected only the bold name segment, no empty rest segment");
133        let DrawOp::Text { text: name, bold: name_bold, .. } = &ops[0] else { panic!("expected Text") };
134        assert_eq!(name, "NoSlashHere");
135        assert!(name_bold);
136    }
137
138    #[test]
139    fn rest_is_truncated_to_fit_available_width() {
140        let (bill, fonts) = ctx_and_fonts();
141        let ctx = RenderContext::for_strategy::<PaymentPartLayout<SvgFontLibrary>>(&bill, Language::De, &fonts);
142
143        let long_rest = "a-very-long-identifier-that-will-not-fit-in-a-narrow-column@example.com";
144        let scheme = format!("eBill/{long_rest}");
145
146        let mut ops = Vec::new();
147        // Deliberately narrow: enough for the bold name, not for the rest.
148        draw_alt_procedure_line(&mut ops, &ctx, &scheme, Mm(0.0), Mm(0.0), Mm(20.0));
149
150        let DrawOp::Text { text: rest, .. } = &ops[1] else { panic!("expected a truncated rest segment") };
151        assert!(rest.ends_with('…'), "expected truncation ellipsis, got {rest:?}");
152        assert!(rest.len() < long_rest.len());
153    }
