7use crate::block_elements::{Column, ColumnCursor, LayoutBlock};
8use crate::constants::{A4_PAGE_HEIGHT, A4_PAGE_WIDTH, MARGIN, RECEIPT_WIDTH, PP_FURTHER_INFO_LINE_SPACING, PP_FURTHER_INFO_SECTION_TOP, PP_FURTHER_INFO_TEXT_SIZE};
9use crate::pdf::coords::LayoutY;
10use crate::render::FontMetrics;
11use crate::{truncate_to_width, Baseline, DrawOp, FontStyle, Mm, RenderContext};
12
13/// Style Guide 3.5.5 "Further information section" / "Weitere
14/// Informationen": up to two "Alternative procedures" lines at the bottom
15/// of the payment part, 7pt, with the (short) procedure name in bold. The
16/// receipt has no such section.
17pub struct FurtherInformationBlock;
18
19impl<T: FontMetrics> LayoutBlock<T> for FurtherInformationBlock {
20    fn column(&self) -> Column {
21        Column::Absolute
22    }
23
24    fn render(&self, ctx: &RenderContext<'_, T>, ops: &mut Vec<DrawOp>, _cursor: &mut ColumnCursor) {
25        let x = RECEIPT_WIDTH + MARGIN;
26        let available_width = A4_PAGE_WIDTH - x - MARGIN;
27        let row_top = A4_PAGE_HEIGHT - PP_FURTHER_INFO_SECTION_TOP
28            + ctx.metrics.ascender_mm(FontStyle::Bold, PP_FURTHER_INFO_TEXT_SIZE);
29
30        for (i, scheme) in ctx.bill_data.alternative_schemes.iter().flatten().enumerate() {
31            let y = row_top + Mm(i as f32) * PP_FURTHER_INFO_LINE_SPACING.to_mm();
32            draw_alt_procedure_line(ops, ctx, scheme, x, y, available_width);
33        }
34    }
35}
36
37/// Draws one Alternative-procedure line: the part up to and including the
38/// first separator ("eBill/" in "eBill/B/simon.muster@example.com") in
39/// bold, the rest in regular weight, truncated with "…" if it would
40/// otherwise overflow `available_width`.
41fn draw_alt_procedure_line<T: FontMetrics>(
42    ops: &mut Vec<DrawOp>,
43    ctx: &RenderContext<'_, T>,
44    scheme: &str,
45    x: Mm,
46    y: Mm,
47    available_width: Mm,
48) {
49    let (name, rest) = match scheme.find('/') {
50        Some(idx) => scheme.split_at(idx + 1),
51        None => (scheme, ""),
52    };
53
54    let name_width = ctx.metrics.text_width_mm(name, FontStyle::Bold, PP_FURTHER_INFO_TEXT_SIZE);
55    let rest = truncate_to_width(
56        ctx.metrics,
57        rest,
58        FontStyle::Regular,
59        PP_FURTHER_INFO_TEXT_SIZE,
60        available_width - name_width,
61    );
62
63    ops.push(DrawOp::Text {
64        text: name.to_string(),
65        at: Baseline { x, y: LayoutY(y) },
66        size: PP_FURTHER_INFO_TEXT_SIZE,
67        bold: true,
68    });
69
70    if !rest.is_empty() {
71        ops.push(DrawOp::Text {
72            text: rest,
73            at: Baseline { x: x + name_width, y: LayoutY(y) },
74            size: PP_FURTHER_INFO_TEXT_SIZE,
75            bold: false,
76        });
77    }
78}
