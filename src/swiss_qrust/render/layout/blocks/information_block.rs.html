6use crate::support::traits::SwissQRFormatter;
7use crate::block_elements::{Column, ColumnCursor, LayoutBlock};
8use crate::constants::{A4_PAGE_WIDTH, MARGIN};
9use crate::{draw_corner_marks, draw_label, draw_single_line, draw_text_lines, label, truncate_to_width, DrawOp, FontStyle, Mm, QRBillLayoutRect, ReferenceType, RenderContext, SlipPart, CORNER_MARKS_PAYABLE_BY_POLYLINES, CORNER_MARKS_PAYABLE_BY_VIEWBOX};
10use crate::render::FontMetrics;
11use crate::support::traits::SliceExt;
12
13pub struct InformationBlock{
14    pub part: SlipPart,
15    pub payable_box_width: Mm,
16    pub payable_box_height: Mm,
17    pub offset: Mm
18}
19impl <T: FontMetrics> LayoutBlock<T> for InformationBlock {
20    fn column(&self) -> Column {
21        Column::Right
22    }
23
24    /// Style Guide p.15: "Da es keine fixen Platzierungen der einzelnen
25    /// Textblöcke gibt, rückt bei fehlenden Textblöcken jeweils alles nach
26    /// oben" - there are no fixed positions for these fields; a block with
27    /// no content (no reference, no message, ...) is omitted entirely (no
28    /// heading, no reserved space) and everything below it moves up. Blocks
29    /// that are present are separated by exactly one blank line (Zeilenabstand
30    /// 9pt on the receipt / 11pt on the payment part, via `ctx.line_spacing`).
31    fn render(&self, ctx: &RenderContext<'_, T>, ops: &mut Vec<DrawOp>, cursor: &mut ColumnCursor) {
32        // Account / Payable to
33        draw_label(ops, label!(AccountPayableTo, ctx.language), cursor.x, &mut cursor.y, ctx.label_size);
34        cursor.advance(ctx.line_spacing);
35        draw_single_line(ops, &ctx.bill_data.iban.format_iban(), cursor.x, &mut cursor.y, ctx.text_size);
36        cursor.advance(ctx.line_spacing);
37        draw_text_lines(ops, &ctx.bill_data.creditor_address.to_lines().all_but_last(), cursor.x, &mut cursor.y, ctx.text_size, ctx.line_spacing);
38        cursor.advance(ctx.line_spacing); // blank line before the next block
39
40        // Reference - omitted entirely (no heading) when the bill has none.
41        let reference_value = match &ctx.bill_data.reference_type {
42            ReferenceType::QrRef(reference) => Some(reference.format_qr_reference()),
43            ReferenceType::Creditor(reference) => Some(reference.format_scor_reference()),
44            ReferenceType::NoRef => None,
45        };
46        if let Some(reference_value) = reference_value {
47            draw_label(ops, label!(Reference, ctx.language), cursor.x, &mut cursor.y, ctx.label_size);
48            cursor.advance(ctx.line_spacing);
49            draw_single_line(ops, &reference_value, cursor.x, &mut cursor.y, ctx.text_size);
50            cursor.advance(ctx.line_spacing); // move past the value line
51            cursor.advance(ctx.line_spacing); // blank line before the next block
52        }
53
54        // Additional information - payment part only; omitted on the
55        // receipt (which has no such field) and when there is nothing to
56        // show. Style Guide 3.5.4: Ustrd and StrdBkgInf (which may hold
57        // Swico billing information) are shown together here, one per line
58        // if both are present, each truncated with "…" if it overflows the
59        // column width.
60        if self.part == SlipPart::PaymentPart {
61            let lines: Vec<&str> = [
62                ctx.bill_data.unstructured_message.as_deref(),
63                ctx.bill_data.bill_information.as_deref(),
64            ]
65            .into_iter()
66            .flatten()
67            .collect();
68
69            if !lines.is_empty() {
70                let available_width = A4_PAGE_WIDTH - cursor.x - MARGIN;
71                draw_label(ops, label!(AdditionalInformation, ctx.language), cursor.x, &mut cursor.y, ctx.label_size);
72                cursor.advance(ctx.line_spacing);
73                for line in lines {
74                    let line = truncate_to_width(ctx.metrics, line, FontStyle::Regular, ctx.text_size, available_width);
75                    draw_single_line(ops, &line, cursor.x, &mut cursor.y, ctx.text_size);
76                    cursor.advance(ctx.line_spacing); // move past the value line
77                }
78                cursor.advance(ctx.line_spacing); // blank line before the next block
79            }
80        }
81
82        // Payable by
83        if let Some(debtor) = &ctx.bill_data.debtor_address {
84            draw_label(ops, label!(PayableBy, ctx.language), cursor.x, &mut cursor.y, ctx.label_size);
85            cursor.advance(ctx.line_spacing);
86            draw_text_lines(ops, &debtor.to_lines().all_but_last(), cursor.x, &mut cursor.y, ctx.text_size, ctx.line_spacing);
87        } else {
88            draw_label(ops, label!(PayableByNameAddress, ctx.language), cursor.x, &mut cursor.y, ctx.label_size);
89            cursor.advance(ctx.line_spacing);
90            draw_corner_marks(ops, QRBillLayoutRect {x: cursor.x, y: cursor.y, width: self.payable_box_width, height: self.payable_box_height  }, CORNER_MARKS_PAYABLE_BY_VIEWBOX, CORNER_MARKS_PAYABLE_BY_POLYLINES);
91            cursor.advance(self.payable_box_height);
92        }
93    }
94}
