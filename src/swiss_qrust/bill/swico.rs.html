7//! Swico S1 "Billing information" (Rechnungsinformationen): the structured
8//! billing information some Swiss accounting software puts into the
9//! QR-bill's `StrdBkgInf` field, per Implementation Guidelines 4.3.3 and
10//! Annex D. Not part of the SIX standardisation itself - Swico publishes
11//! and owns the S1 syntax (<http://swiss-qr-invoice.org/downloads/qr-bill-s1-syntax-de.pdf>).
12use chrono::{Days, NaiveDate};
13use crate::BillError;
14use crate::input::InputSwicoBillInformation;
15
16#[derive(Debug, Default, Clone, PartialEq)]
17pub struct SwicoBillInformation {
18    pub invoice_number: Option<String>,
19    pub invoice_date: Option<NaiveDate>,
20    pub customer_reference: Option<String>,
21    pub vat_number: Option<String>,
22    /// Date the goods/service were supplied. Mutually exclusive with
23    /// `vat_start_date` / `vat_end_date`.
24    pub vat_date: Option<NaiveDate>,
25    pub vat_start_date: Option<NaiveDate>,
26    pub vat_end_date: Option<NaiveDate>,
27    /// VAT rate (percent) if a single rate applies to the whole invoice.
28    /// Mutually exclusive with `vat_rate_details`.
29    pub vat_rate: Option<f64>,
30    /// VAT rate/net-amount tuples, if different rates apply to different
31    /// line items.
32    pub vat_rate_details: Vec<RateDetail>,
33    /// VAT rate/tax-amount tuples for imported goods.
34    pub vat_import_taxes: Vec<RateDetail>,
35    /// Discount/deadline-in-days tuples (e.g. 2% within 10 days, due in 30).
36    pub payment_conditions: Vec<PaymentCondition>,
37}
38
39#[derive(Debug, Clone, Copy, PartialEq)]
40pub struct RateDetail {
41    pub rate: f64,
42    pub amount: f64,
43}
44
45#[derive(Debug, Clone, Copy, PartialEq)]
46pub struct PaymentCondition {
47    pub discount: f64,
48    pub days: u32,
49}

51impl SwicoBillInformation {
52    /// Encodes this bill information as `StrdBkgInf` text (Swico S1
53    /// syntax), or `None` if nothing is set.
54    pub fn encode_as_text(&self) -> Option<String> {
55        let mut out = String::from("//S1");
56
57        if let Some(v) = &self.invoice_number {
58            out.push_str("/10/");
59            out.push_str(&escape(v));
60        }
61        if let Some(d) = self.invoice_date {
62            out.push_str("/11/");
63            out.push_str(&s1_date(d));
64        }
65        if let Some(v) = &self.customer_reference {
66            out.push_str("/20/");
67            out.push_str(&escape(v));
68        }
69        if let Some(v) = &self.vat_number {
70            out.push_str("/30/");
71            out.push_str(&escape(v));
72        }
73
74        if let Some(d) = self.vat_date {
75            out.push_str("/31/");
76            out.push_str(&s1_date(d));
77        } else if let (Some(start), Some(end)) = (self.vat_start_date, self.vat_end_date) {
78            out.push_str("/31/");
79            out.push_str(&s1_date(start));
80            out.push_str(&s1_date(end));
81        }
82
83        if let Some(rate) = self.vat_rate {
84            out.push_str("/32/");
85            out.push_str(&s1_number(rate));
86        } else if !self.vat_rate_details.is_empty() {
87            out.push_str("/32/");
88            out.push_str(&rate_detail_list(&self.vat_rate_details));
89        }
90
91        if !self.vat_import_taxes.is_empty() {
92            out.push_str("/33/");
93            out.push_str(&rate_detail_list(&self.vat_import_taxes));
94        }
95
96        if !self.payment_conditions.is_empty() {
97            out.push_str("/40/");
98            out.push_str(&condition_list(&self.payment_conditions));
99        }
100
101        (out.len() > "//S1".len()).then_some(out)
102    }
103
104    /// Decodes `StrdBkgInf` text encoded per Swico S1 syntax. As much data
105    /// as possible is decoded; malformed tuples are silently skipped
106    /// (matching the reference decoder). Returns `None` if `text` isn't a
107    /// Swico S1 payload at all.
108    pub fn decode_text(text: &str) -> Option<Self> {
109        let body = text.strip_prefix("//S1/")?;
110        let parts = split_unescaped(body);
111
112        let mut info = SwicoBillInformation::default();
113        for pair in parts.chunks_exact(2) {
114            let (tag, value) = (pair[0].as_str(), pair[1].as_str());
115            if value.is_empty() {
116                continue;
117            }
118            if let Ok(tag) = tag.parse::<u32>() {
119                decode_element(&mut info, tag, value);
120            }
121        }
122        Some(info)
123    }
124
125    /// The invoice date plus the number of days of the payment condition
126    /// with a 0% discount (the "net" deadline), if both are present.
127    pub fn due_date(&self) -> Option<NaiveDate> {
128        let invoice_date = self.invoice_date?;
129        let net_condition = self.payment_conditions.iter().find(|c| c.discount == 0.0)?;
130        invoice_date.checked_add_days(Days::new(net_condition.days as u64))
131    }
132}

134impl TryFrom<InputSwicoBillInformation> for SwicoBillInformation {
135    type Error = BillError;
136
137    fn try_from(input: InputSwicoBillInformation) -> Result<Self, Self::Error> {
138        Ok(SwicoBillInformation {
139            invoice_number: input.invoice_number,
140            invoice_date: parse_input_date(input.invoice_date)?,
141            customer_reference: input.customer_reference,
142            vat_number: input.vat_number,
143            vat_date: parse_input_date(input.vat_date)?,
144            vat_start_date: parse_input_date(input.vat_start_date)?,
145            vat_end_date: parse_input_date(input.vat_end_date)?,
146            vat_rate: input.vat_rate,
147            vat_rate_details: input
148                .vat_rate_details
149                .unwrap_or_default()
150                .into_iter()
151                .map(|d| RateDetail { rate: d.rate, amount: d.amount })
152                .collect(),
153            vat_import_taxes: input
154                .vat_import_taxes
155                .unwrap_or_default()
156                .into_iter()
157                .map(|d| RateDetail { rate: d.rate, amount: d.amount })
158                .collect(),
159            payment_conditions: input
160                .payment_conditions
161                .unwrap_or_default()
162                .into_iter()
163                .map(|c| PaymentCondition { discount: c.discount, days: c.days })
164                .collect(),
165        })
166    }
167}

169/// Parses an optional "YYYY-MM-DD" input date.
170fn parse_input_date(date: Option<String>) -> Result<Option<NaiveDate>, BillError> {
171    date.map(|s| {
172        NaiveDate::parse_from_str(&s, "%Y-%m-%d").map_err(|_| BillError::InvalidSwicoDate(s.clone()))
173    })
174    .transpose()
175}
176
177fn decode_element(info: &mut SwicoBillInformation, tag: u32, value: &str) {
178    match tag {
179        10 => info.invoice_number = Some(value.to_string()),
180        11 => info.invoice_date = parse_s1_date(value),
181        20 => info.customer_reference = Some(value.to_string()),
182        30 => info.vat_number = Some(value.to_string()),
183        31 => set_vat_dates(info, value),
184        32 => set_vat_rate(info, value),
185        33 => info.vat_import_taxes = parse_rate_detail_list(value),
186        40 => info.payment_conditions = parse_condition_list(value),
187        _ => {} // unknown tags are ignored, per spec Annex D
188    }
189}
190
191fn set_vat_dates(info: &mut SwicoBillInformation, value: &str) {
192    match value.len() {
193        6 => {
194            if let Some(date) = parse_s1_date(value) {
195                info.vat_date = Some(date);
196                info.vat_start_date = None;
197                info.vat_end_date = None;
198            }
199        }
200        12 => {
201            let (start, end) = value.split_at(6);
202            if let (Some(start), Some(end)) = (parse_s1_date(start), parse_s1_date(end)) {
203                info.vat_start_date = Some(start);
204                info.vat_end_date = Some(end);
205                info.vat_date = None;
206            }
207        }
208        _ => {}
209    }
210}
211
212fn set_vat_rate(info: &mut SwicoBillInformation, value: &str) {
213    if !value.contains(':') && !value.contains(';') {
214        if let Some(rate) = parse_decimal(value) {
215            info.vat_rate = Some(rate);
216            info.vat_rate_details = Vec::new();
217        }
218    } else {
219        info.vat_rate_details = parse_rate_detail_list(value);
220        info.vat_rate = None;
221    }
222}
223
224fn parse_rate_detail_list(value: &str) -> Vec<RateDetail> {
225    value
226        .split(';')
227        .filter_map(|entry| {
228            let (rate, amount) = entry.split_once(':')?;
229            Some(RateDetail { rate: parse_decimal(rate)?, amount: parse_decimal(amount)? })
230        })
231        .collect()
232}
233
234fn parse_condition_list(value: &str) -> Vec<PaymentCondition> {
235    value
236        .split(';')
237        .filter_map(|entry| {
238            let (discount, days) = entry.split_once(':')?;
239            Some(PaymentCondition { discount: parse_decimal(discount)?, days: days.parse().ok()? })
240        })
241        .collect()
242}
243
244fn rate_detail_list(list: &[RateDetail]) -> String {
245    list.iter()
246        .map(|d| format!("{}:{}", s1_number(d.rate), s1_number(d.amount)))
247        .collect::<Vec<_>>()
248        .join(";")
249}
250
251fn condition_list(list: &[PaymentCondition]) -> String {
252    list.iter()
253        .map(|c| format!("{}:{}", s1_number(c.discount), c.days))
254        .collect::<Vec<_>>()
255        .join(";")
256}
257
258fn escape(text: &str) -> String {
259    text.replace('\\', "\\\\").replace('/', "\\/")
260}
261
262/// Splits `text` at unescaped slashes, undoing the backslash escaping.
263/// Mirrors the reference decoder's placeholder trick: `\\` and `\/` are
264/// swapped for characters outside the QR-bill character set before
265/// splitting, so an escaped slash never looks like a separator.
266fn split_unescaped(text: &str) -> Vec<String> {
267    text.replace("\\\\", "\u{2601}")
268        .replace("\\/", "\u{2605}")
269        .split('/')
270        .map(|s| s.replace('\u{2605}', "/").replace('\u{2601}', "\\"))
271        .collect()
272}
273
274fn s1_date(date: NaiveDate) -> String {
275    date.format("%y%m%d").to_string()
276}
277
278/// Parses a Swico S1 date. 6 digits (yyMMdd) per spec; 10 and 12 digits
279/// (with a time-of-day suffix) are accepted too - not per spec, but seen in
280/// production, per the reference implementation.
281fn parse_s1_date(text: &str) -> Option<NaiveDate> {
282    match text.len() {
283        6 => NaiveDate::parse_from_str(text, "%y%m%d").ok(),
284        10 => NaiveDate::parse_from_str(text, "%y%m%d%H%M").ok(),
285        12 => NaiveDate::parse_from_str(text, "%y%m%d%H%M%S").ok(),
286        _ => None,
287    }
288}
289
290/// Formats like Java's `DecimalFormat("0.###")`: up to 3 decimal places,
291/// trailing zeros (and a bare trailing dot) dropped.
292fn s1_number(n: f64) -> String {
293    let rounded = (n * 1000.0).round() / 1000.0;
294    let mut s = format!("{:.3}", rounded);
295    while s.ends_with('0') {
296        s.pop();
297    }
298    if s.ends_with('.') {
299        s.pop();
300    }
301    if s.is_empty() || s == "-" {
302        s = "0".to_string();
303    }
304    s
305}
306
307fn parse_decimal(s: &str) -> Option<f64> {
308    s.parse::<f64>().ok()
309}

323    fn decodes_spec_example_1() {
324        let info = SwicoBillInformation::decode_text(
325            "//S1/10/10201409/11/190512/20/1400.000-53/30/106017086/31/180508/32/7.7/40/2:10;0:30",
326        )
327        .unwrap();
328
329        assert_eq!(info.invoice_number.as_deref(), Some("10201409"));
330        assert_eq!(info.invoice_date, Some(date(2019, 5, 12)));
331        assert_eq!(info.customer_reference.as_deref(), Some("1400.000-53"));
332        assert_eq!(info.vat_number.as_deref(), Some("106017086"));
333        assert_eq!(info.vat_date, Some(date(2018, 5, 8)));
334        assert_eq!(info.vat_rate, Some(7.7));

339                PaymentCondition { discount: 0.0, days: 30 },

398        let original = SwicoBillInformation {
399            invoice_number: Some("X.66711/8824".to_string()),

411        let encoded = original.encode_as_text().unwrap();
